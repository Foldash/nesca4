cmake_minimum_required(VERSION 3.16)
project(nesca4 VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/src")
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets)

# This target deliberately has no network engine or legacy background threads.
add_executable(nesca4-ui tools/ui_preview.cpp src/nesca_3.ui src/ModernTheme.h)
target_include_directories(nesca4-ui PRIVATE src)
target_link_libraries(nesca4-ui PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets)
if(MSVC)
    target_compile_options(nesca4-ui PRIVATE /W4 /utf-8)
else()
    target_compile_options(nesca4-ui PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_test(NAME ui-smoke COMMAND nesca4-ui --smoke)
    set_tests_properties(ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(utils-regression tests/utils_regression.cpp)
    target_include_directories(utils-regression PRIVATE src)
    target_link_libraries(utils-regression PRIVATE Qt5::Core)
    add_test(NAME utils-regression COMMAND utils-regression)
    add_executable(bounded-response tests/bounded_response.cpp)
    target_include_directories(bounded-response PRIVATE src)
    add_test(NAME bounded-response COMMAND bounded-response)
endif()
