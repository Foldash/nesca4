cmake_minimum_required(VERSION 3.16)
project(nesca4 VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/src")
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets)
find_package(Threads REQUIRED)

option(NESCA_BUILD_LEGACY "Build the legacy application for engineering validation" OFF)

add_library(nesca4-runtime STATIC src/ThreadPool.cpp src/ThreadPool.h src/TargetList.cpp src/TargetList.h)
target_include_directories(nesca4-runtime PUBLIC src)
target_link_libraries(nesca4-runtime PUBLIC Threads::Threads)

# This target deliberately has no network engine or legacy background threads.
add_executable(nesca4-ui tools/ui_preview.cpp src/nesca_3.ui src/ModernTheme.h)
target_include_directories(nesca4-ui PRIVATE src)
target_link_libraries(nesca4-ui PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets)
if(WIN32)
    set_target_properties(nesca4-ui PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(nesca4-ui PRIVATE Qt5::WinMain)
endif()

if(NESCA_BUILD_LEGACY)
    find_package(OpenSSL 1.1.1 REQUIRED)
    find_package(CURL REQUIRED)
    find_path(LIBSSH_INCLUDE_DIR libssh/libssh.h)
    find_library(LIBSSH_LIBRARY NAMES ssh)
    if(NOT LIBSSH_INCLUDE_DIR OR NOT LIBSSH_LIBRARY)
        message(FATAL_ERROR "The legacy engineering build requires libssh development files")
    endif()
    file(GLOB LEGACY_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
    list(REMOVE_ITEM LEGACY_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/ThreadPool.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/TargetList.cpp")
    add_executable(nesca4-legacy ${LEGACY_SOURCES} src/nesca_3.ui)
    set_target_properties(nesca4-legacy PROPERTIES AUTOMOC ON)
    target_include_directories(nesca4-legacy PRIVATE src "${LIBSSH_INCLUDE_DIR}")
    target_link_libraries(nesca4-legacy PRIVATE nesca4-runtime Qt5::Core Qt5::Gui Qt5::Widgets
        OpenSSL::SSL OpenSSL::Crypto CURL::libcurl "${LIBSSH_LIBRARY}")
    if(MSVC)
        target_compile_options(nesca4-legacy PRIVATE /utf-8)
        target_link_libraries(nesca4-legacy PRIVATE ws2_32 iphlpapi)
    endif()
endif()
if(MSVC)
    target_compile_options(nesca4-ui PRIVATE /W4 /utf-8)
else()
    target_compile_options(nesca4-ui PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(target-list-regression tests/target_list_regression.cpp)
    target_link_libraries(target-list-regression PRIVATE nesca4-runtime)
    add_test(NAME target-list-regression COMMAND target-list-regression)
    set_tests_properties(target-list-regression PROPERTIES TIMEOUT 10)
    add_test(NAME ui-smoke COMMAND nesca4-ui --smoke)
    set_tests_properties(ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(utils-regression tests/utils_regression.cpp)
    target_include_directories(utils-regression PRIVATE src)
    target_link_libraries(utils-regression PRIVATE Qt5::Core)
    add_test(NAME utils-regression COMMAND utils-regression)
    add_executable(bounded-response tests/bounded_response.cpp)
    target_include_directories(bounded-response PRIVATE src)
    add_test(NAME bounded-response COMMAND bounded-response)
    add_executable(thread-pool-regression tests/thread_pool_regression.cpp)
    target_link_libraries(thread-pool-regression PRIVATE nesca4-runtime)
    if(NESCA_BUILD_LEGACY)
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/legacy-check")
        add_test(NAME legacy-ui-check COMMAND nesca4-legacy --ui-check)
        add_test(NAME legacy-session-check COMMAND nesca4-legacy --session-check)
        add_test(NAME legacy-import-check COMMAND nesca4-legacy --import-check)
        set_tests_properties(legacy-ui-check legacy-session-check legacy-import-check PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
            WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/legacy-check" TIMEOUT 10)
    endif()
    add_test(NAME thread-pool-regression COMMAND thread-pool-regression)
    set_tests_properties(thread-pool-regression PROPERTIES TIMEOUT 20)
    add_executable(threader-integration tests/threader_integration.cpp src/Threader.cpp)
    target_link_libraries(threader-integration PRIVATE nesca4-runtime Qt5::Core)
    add_test(NAME threader-integration COMMAND threader-integration)
    set_tests_properties(threader-integration PROPERTIES TIMEOUT 10)
endif()
